\documentclass{amsart}
\begin{document}

\subsection*{Problem}
Solve for $x$:
\[ 9^x - 3^{\,x+1} = 54 \]

\subsection*{Solution}
Let $t = 3^x$, so $9^x = t^2$
and $3^{\,x+1} = 3t$:
\[ t^2 - 3t - 54 = 0 \]
\[ (t - 9)(t + 6) = 0 \]
Since $3^x > 0$, reject $t = -6$.
\[ 3^x = 9 \implies \boxed{x = 2} \]

\end{document}
